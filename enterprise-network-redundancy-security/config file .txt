1. ISP Router (R1)

en
conf t
interface g0/0
 ip address 203.0.32.1 255.255.255.252
 no shutdown
exit
interface g0/1
 ip address 203.0.32.5 255.255.255.252
 no shutdown
exit
ip route 192.168.32.0 255.255.255.0 203.0.32.3
ip route 192.168.32.0 255.255.255.0 203.0.32.6

2. Main Router (R2)

en
conf t
no interface Port-channel 1
interface Port-channel 1
 ip address 192.168.32.1 255.255.255.0
 ip nat inside
 standby 1 ip 192.168.32.100
 standby 1 priority 110
 standby 1 preempt
 no shutdown
exit
interface g0/0
 ip address 203.0.32.3 255.255.255.252
 ip nat outside
 no shutdown
exit
interface range f0/1/0 - 1
 no ip address
 channel-group 1 mode on
 no shutdown
exit
ip nat inside source list 1 interface g0/0 overload
access-list 1 permit 192.168.32.0 0.0.0.255
ip route 0.0.0.0 0.0.0.0 203.0.32.1

4. Core Switch (SW1)

en
conf t
no interface Port-channel 1
no interface Port-channel 2
interface Port-channel 1
 switchport mode access
 switchport access vlan 1
 spanning-tree portfast
exit
interface Port-channel 2
 switchport mode access
 switchport access vlan 1
 spanning-tree portfast
exit
interface range g0/1, g1/1
 switchport mode access
 switchport access vlan 1
 channel-group 1 mode on
 no shutdown
exit
interface range g2/1, g3/1
 switchport mode access
 switchport access vlan 1
 channel-group 2 mode on
 no shutdown
exit
interface f9/1
 switchport mode trunk
 switchport trunk allowed vlan 1-1005
 no shutdown
exit
